# P0-8 Agent 循环 Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** 实现手写 ReAct function-calling Agent 循环：LLM 按需调用工具 → 追加结果 → 循环直到最终答案或达到最大迭代次数，并通过 SSE `step` 事件推送工具调用过程。

**Architecture:** runtime 层新增 `run_agent()` 编排函数；通过 `models.invoke_with_tools()` 调用支持 function-calling 的 LLM；工具执行委托给 `tools.call_tool()`；agent apps 独立端点 `POST /apps/{id}/run`；前端 AgentPage 展示 step 事件气泡。

**Tech Stack:** FastAPI · LiteLLM function calling · SQLAlchemy 2.0 · React + Ant Design + TanStack Query · SSE

## Global Constraints

- 后端命令：`wsl -d Ubuntu-26.04 bash -c "cd /mnt/e/codespace/project/me/agent-hify/backend && <cmd>"`
- Agent 循环上限：`max_iterations`（默认 10），超限后直接发最后一次 LLM 回复
- import-linter：移除 runtime 的 tools 禁止项（runtime 此后可以依赖 tools）
- 工具调用单步失败不中断循环，工具结果以 `is_error` 标记后继续
- SSE 事件序列：`step*` → `message*` → `usage` → `done`
- Agent app 终止响应后保存 content blocks（含每步 tool_use/tool_result）

```python
async def invoke_with_tools(
    ref: ModelRef,
    messages: list[dict[str, Any]],
    tools: list[dict[str, Any]] | None = None,
    extra_params: dict[str, Any] | None = None,
) -> InvokeResult:
    async def _call() -> InvokeResult:
        kwargs: dict[str, Any] = {
            "model": ref.model_key,
            "messages": messages,
            "api_key": ref.api_key,
            "api_base": ref.base_url,
            **ref.default_params,
            **(extra_params or {}),
        }
        if tools:
            kwargs["tools"] = tools
            kwargs["tool_choice"] = "auto"
        resp = await litellm.acompletion(**kwargs)
        choice = resp.choices[0]
        usage = resp.usage
        try:
            cost = float(litellm.completion_cost(completion_response=resp))
        except Exception:
            cost = 0.0
        parsed_tool_calls: list[dict[str, Any]] | None = None
        if choice.message.tool_calls:
            parsed_tool_calls = [
                {
                    "id": tc.id,
                    "type": tc.type,
                    "function": {
                        "name": tc.function.name,
                        "arguments": tc.function.arguments,
                    },
                }
                for tc in choice.message.tool_calls
            ]
        return InvokeResult(
            content=choice.message.content or "",
            tokens_in=int(usage.prompt_tokens),
            tokens_out=int(usage.completion_tokens),
            cost=cost,
            finish_reason=str(choice.finish_reason or "stop"),
            tool_calls=parsed_tool_calls,
        )

    return await call_with_resilience(
        f"{ref.provider_id}:invoke_tools", _call, timeout=_TIMEOUT, retryable=_RETRYABLE
    )
```

```python
def _find_tool_id_by_name(
    session: Session,
    name: str,
    workspace_id: int,
    tool_ids: list[int],
) -> int | None:
    for tid in tool_ids:
        try:
            out = tools_service.get_tool_out(session, tid, workspace_id)
            if out.name == name:
                return tid
        except Exception:
            continue
    return None


async def run_agent(
    app_id: int,
    workspace_id: int,
    user_id: int,
    payload: ChatInput,
) -> AsyncGenerator[str, None]:
    """ReAct function-calling Agent 循环。"""
    from agent_hify.modules.apps.schemas import AppConfigAgent

    session = SessionLocal()
    try:
        app = apps_service.get_app(session, app_id, workspace_id)
        if app.type != "agent":
            raise ValidationError(ErrorCode.PARAM_INVALID, "该应用不是 Agent 类型")
        config = AppConfigAgent.model_validate(app.config)

        if payload.conversation_id is not None:
            conv = repository.get_conversation(session, payload.conversation_id, workspace_id)
            if conv is None:
                raise NotFoundError(ErrorCode.CONVERSATION_NOT_FOUND, "对话不存在")
        else:
            conv = repository.insert_conversation(
                session,
                Conversation(
                    app_id=app_id,
                    workspace_id=workspace_id,
                    user_id=user_id,
                    title=_title_from(payload.message),
                ),
            )

        repository.insert_message(
            session,
            Message(
                conversation_id=conv.id,
                role="user",
                content=[{"type": "text", "text": payload.message}],
            ),
        )

        history = repository.get_recent_messages(session, conv.id, config.history_limit)
        llm_messages: list[dict[str, object]] = []
        if config.system_prompt:
            llm_messages.append({"role": "system", "content": config.system_prompt})
        for m in history:
            llm_messages.append({"role": m.role, "content": _text_of(m.content)})

        tool_specs = tools_service.get_specs(session, list(config.tool_ids), workspace_id)
        tools_json: list[dict[str, object]] = [
            {
                "type": "function",
                "function": {
                    "name": spec.name,
                    "description": spec.description,
                    "parameters": spec.parameters,
                },
            }
            for spec in tool_specs
        ]

        extra_params: dict[str, object] = dict(config.params)
        final_answer = ""
        last_result = None
        started = time.monotonic()

        for iteration in range(config.max_iterations):
            result = await models_service.invoke_with_tools(
                session,
                model_id=config.model_id,
                workspace_id=workspace_id,
                messages=list(llm_messages),
                tools=list(tools_json) if tools_json else None,
                extra_params=extra_params,
            )
            last_result = result

            if result.tool_calls:
                asst_msg: dict[str, object] = {
                    "role": "assistant",
                    "content": result.content or "",
                    "tool_calls": list(result.tool_calls),
                }
                llm_messages.append(asst_msg)

                for tc in result.tool_calls:
                    tool_name = str(tc["function"]["name"])  # type: ignore[index]
                    try:
                        import json as _json
                        args = _json.loads(str(tc["function"]["arguments"]))  # type: ignore[index]
                    except Exception:
                        args = {}

                    tool_id = _find_tool_id_by_name(
                        session, tool_name, workspace_id, list(config.tool_ids)
                    )
                    if tool_id is None:
                        tool_result_content = f"Tool '{tool_name}' not found in app config"
                        is_error = True
                    else:
                        tr = await tools_service.call_tool(session, tool_id, workspace_id, args)
                        tool_result_content = tr.content
                        is_error = tr.is_error

                    yield _sse("step", {
                        "iteration": iteration + 1,
                        "tool_name": tool_name,
                        "tool_args": args,
                        "tool_result": tool_result_content,
                        "is_error": is_error,
                    })

                    obs_service.record_trace(
                        session,
                        TraceIn(
                            workspace_id=workspace_id,
                            type="tool_call",
                            status="error" if is_error else "ok",
                            app_id=app_id,
                            conversation_id=conv.id,
                            input={"tool_name": tool_name, "args": args},
                            output={"content": tool_result_content},
                        ),
                    )

                    llm_messages.append({
                        "role": "tool",
                        "tool_call_id": str(tc["id"]),  # type: ignore[index]
                        "content": tool_result_content,
                    })
            else:
                final_answer = result.content
                break
        else:
            final_answer = last_result.content if last_result else "（已达最大迭代次数）"

        latency_ms = int((time.monotonic() - started) * 1000)

        yield _sse("message", {"delta": final_answer})

        assistant = repository.insert_message(
            session,
            Message(
                conversation_id=conv.id,
                role="assistant",
                content=[{"type": "text", "text": final_answer}],
            ),
        )

        tokens_in = int(str((last_result.tokens_in if last_result else 0)))
        tokens_out = int(str((last_result.tokens_out if last_result else 0)))
        cost = Decimal(str(last_result.cost if last_result else 0))

        obs_service.record_trace(
            session,
            TraceIn(
                workspace_id=workspace_id,
                type="llm_call",
                status="ok",
                app_id=app_id,
                conversation_id=conv.id,
                message_id=assistant.id,
                tokens_in=tokens_in,
                tokens_out=tokens_out,
                cost=cost,
                latency_ms=latency_ms,
                input={"messages": list(llm_messages)},
                output={"content": final_answer},
            ),
        )

        yield _sse("usage", {
            "tokens_in": tokens_in,
            "tokens_out": tokens_out,
            "cost": float(cost),
        })
        yield _sse("done", {"conversation_id": conv.id, "message_id": assistant.id})
        session.commit()
    except AppError as exc:
        yield _sse("error", {"code": exc.code, "message": exc.message})
    except Exception as exc:
        yield _sse("error", {"code": 19001, "message": str(exc)})
    finally:
        session.close()
```

复用 `chat/api.ts` 中的 `streamChat` 函数模式，新增 `runAgent()` 支持 `onStep` handler：

```typescript
// frontend/src/features/agent/api.ts
import { getToken } from '@/lib/auth';

export interface StepEvent {
  iteration: number;
  tool_name: string;
  tool_args: Record<string, unknown>;
  tool_result: string;
  is_error: boolean;
}

export interface AgentHandlers {
  onStep?: (step: StepEvent) => void;
  onDelta?: (delta: string) => void;
  onUsage?: (usage: { tokens_in: number; tokens_out: number; cost: number }) => void;
  onDone?: (data: { conversation_id: number; message_id: number }) => void;
  onError?: (message: string) => void;
}

export async function runAgent(
  appId: number,
  body: { message: string; conversation_id?: number },
  handlers: AgentHandlers,
): Promise<void> {
  const token = getToken();
  const resp = await fetch(`/api/v1/apps/${appId}/run`, {
    method: 'POST',
    headers: {
      'Content-Type': 'application/json',
      ...(token ? { Authorization: `Bearer ${token}` } : {}),
    },
    body: JSON.stringify(body),
  });

  if (!resp.ok) {
    handlers.onError?.(`请求失败：${resp.status}`);
    return;
  }

  const reader = resp.body?.getReader();
  if (!reader) return;
  const decoder = new TextDecoder();
  let buffer = '';

  while (true) {
    const { done, value } = await reader.read();
    if (done) break;
    buffer += decoder.decode(value, { stream: true });
    const blocks = buffer.split('\n\n');
    buffer = blocks.pop() ?? '';
    for (const block of blocks) {
      const eventMatch = block.match(/^event: (\w+)/m);
      const dataMatch = block.match(/^data: (.+)/m);
      if (!eventMatch || !dataMatch) continue;
      const eventType = eventMatch[1];
      try {
        const payload = JSON.parse(dataMatch[1]);
        if (eventType === 'step') handlers.onStep?.(payload as StepEvent);
        else if (eventType === 'message') handlers.onDelta?.(payload.delta as string);
        else if (eventType === 'usage') handlers.onUsage?.(payload);
        else if (eventType === 'done') handlers.onDone?.(payload);
        else if (eventType === 'error') handlers.onError?.(payload.message as string);
      } catch {
        // ignore parse errors
      }
    }
  }
}
```

```tsx
  const [input, setInput] = useState('');
  const [streaming, setStreaming] = useState(false);
  const [convId, setConvId] = useState<number | undefined>();
  const bottomRef = useRef<HTMLDivElement>(null);

  const append = (item: MsgItem) => setMsgs((prev) => [...prev, item]);

  const send = async () => {
    if (!input.trim() || streaming || !appId) return;
    const text = input.trim();
    setInput('');
    setStreaming(true);
    append({ kind: 'user', text });
    let assistantText = '';
    const assistantIdx = { current: -1 };

    try {
      await runAgent(Number(appId), { message: text, conversation_id: convId }, {
        onStep: (step) => append({ kind: 'step', step }),
        onDelta: (delta) => {
          assistantText += delta;
          setMsgs((prev) => {
            const copy = [...prev];
            if (assistantIdx.current === -1) {
              assistantIdx.current = copy.length;
              copy.push({ kind: 'assistant', text: assistantText });
            } else {
              copy[assistantIdx.current] = { kind: 'assistant', text: assistantText };
            }
            return copy;
          });
        },
        onDone: (d) => setConvId(d.conversation_id),
        onError: (msg) => message.error(msg || '请求失败，请重试'),
      });
    } finally {
      setStreaming(false);
    }
    bottomRef.current?.scrollIntoView({ behavior: 'smooth' });
  };

  return (
    <Content style={{ padding: 24, display: 'flex', flexDirection: 'column', height: '100%' }}>
      <div style={{ flex: 1, overflowY: 'auto', marginBottom: 16 }}>
        {msgs.map((m, i) => {
          if (m.kind === 'user') {
            return (
              <div key={i} style={{ textAlign: 'right', marginBottom: 12 }}>
                <Tag color="blue">{m.text}</Tag>
              </div>
            );
          }
          if (m.kind === 'step') {
            return (
              <div key={i} style={{ marginBottom: 8 }}>
                <Card size="small" style={{ background: '#f6ffed', borderColor: '#b7eb8f' }}>
                  <Space>
                    <Badge color={m.step.is_error ? 'red' : 'green'} />
                    <Text type="secondary">
                      [{m.step.iteration}] {m.step.tool_name}
                    </Text>
                    <Text code style={{ fontSize: 11 }}>
                      {m.step.tool_result.slice(0, 80)}
                    </Text>
                  </Space>
                </Card>
              </div>
            );
          }
          return (
            <div key={i} style={{ marginBottom: 12 }}>
              <Tag>{m.text || <Spin size="small" />}</Tag>
            </div>
          );
        })}
        <div ref={bottomRef} />
      </div>

      <Space.Compact style={{ width: '100%' }}>
        <Input
          value={input}
          onChange={(e) => setInput(e.target.value)}
          onPressEnter={send}
          disabled={streaming}
          placeholder="输入消息，按 Enter 发送…"
        />
        <Button type="primary" icon={<SendOutlined />} loading={streaming} onClick={send} />
      </Space.Compact>
    </Content>
  );
}
```
